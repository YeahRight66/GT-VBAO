#include "common_ps_packdepth.h"

sampler WPDepth             : register( s0 ); // Full-resolution depth
sampler TargetTex           : register( s1 ); // Low-resolution target (RGB: Tinted Color, A: AO)
sampler DepthTex            : register( s2 ); // Low-resolution depth
const float2 TargetSize     : register( c5 ); // Low-resolution texel size {1/rt_w, 1/rt_h}

struct PS_IN
{
    float2 position : VPOS;
    float2 texcoord : TEXCOORD0;
};

half4 main(PS_IN I) : COLOR0 {
    float2 tc = I.texcoord;
    
    float4 low_res_00 = tex2D(TargetTex, tc);
    float4 low_res_10 = tex2D(TargetTex, tc + float2(TargetSize.x, 0));
    float4 low_res_01 = tex2D(TargetTex, tc + float2(0, TargetSize.y));
    float4 low_res_11 = tex2D(TargetTex, tc + TargetSize);
    
    // Sample low-resolution depth from the half-resolution depth buffer
    half4 low_res_viewz = 1/float4(
        tex2D(DepthTex, tc).r,
        tex2D(DepthTex, tc + float2(TargetSize.x, 0)).r,
        tex2D(DepthTex, tc + float2(0, TargetSize.y)).r,
        tex2D(DepthTex, tc + TargetSize).r
    );
    
    // Sample full-resolution depth
    float full_res_viewz = tex2D(WPDepth, tc).a;
    full_res_viewz = 1/UnpackDepth24(PackDepth24(full_res_viewz));
    
    // Bilinear weights
    float2 f = frac(I.texcoord / TargetSize);
    float w_00 = (1.0f - f.x) * (1.0f - f.y);
    float w_10 = f.x * (1.0f - f.y);
    float w_01 = (1.0f - f.x) * f.y;
    float w_11 = f.x * f.y;

    float4 bilat_weights = float4(w_00, w_10, w_01, w_11) / (abs(full_res_viewz.xxxx - low_res_viewz) + 1e-4);
    float total_weight = dot(bilat_weights, 1.0);

    // Bilaterally upscale final tinted color
    float3 color_00 = low_res_00.rgb;
    float3 color_10 = low_res_10.rgb;
    float3 color_01 = low_res_01.rgb;
    float3 color_11 = low_res_11.rgb;

    float3 finalColor = (color_00 * bilat_weights.x +
                         color_10 * bilat_weights.y +
                         color_01 * bilat_weights.z +
                         color_11 * bilat_weights.w) / total_weight;

    return half4(finalColor, 1.0f);
}
